جــمــهــوريــة الــعــراق - وزارة الــتــربــيــة

مــتــوســطــة الــجــواديــن لــلــبــنــيــن

اســتــمــارة تــســجــيــل بــيــانــات الــطــالــب الــإلــكــتــرونــيــة

إرشـادات وتـعـلـيـمـات مـلـء الـاسـتـمـارة:

  • يـرجـى نـقـل ومـلـء جـمـيـع الـبـيـانـات بـدّقـة تـامـة مـطـابـقـة لـمـا هـو مـثـبـت فـي الـبـطـاقـة الـوطـنـيـة الـمـوحـدة الـخـاصـة بـالـطـالـب.
  • تـأكـد مـن كـتـابـة الـاسـم الـكـامـل مـع الـلـقـب (5 كـلـمـات عـلـى الـأقـل) وأرقـام الـهـواتـف بـشـكـل صـحـيـح وخـالٍ مـن أي مـسـافـات زائـدة أو رمـوز قـبـل الـإرسـال.
  • يـمـنـع مـنـعـاً بـاتـاً تـكـرار تـقـديـم الـاسـتـمـارة لـنـفـس الـطـالـب، حـيـث سـيـكـشـف الـنـظـام الـتـكـرار تـلـقـائـيـاً ويـقـوم بـإلـغـاء الـطـلـب.
  • تـأكـد مـن اخـتـيـار الـصـف الـدراسـي والـشـعـبـة بـدّقـة لـضـمـان تـسـجـيـل الـبـيـانـات فـي سـجـل الـمـدروسـة الـصـحـيـح.
  • مـراجـعـة حـقـول عـنـوان الـسـكـن (الـمـحـلـة، الـزقـاق، الـدار) وكـتـابـتـهـا بـالـأرقـام فـقـط لـضـمـان سـهـولـة الـوصـول لـلـبـيـانـات عـنـد الـحـاجـة.
